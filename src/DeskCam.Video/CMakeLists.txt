cmake_minimum_required(VERSION 3.20)
project(DeskCamVideo LANGUAGES CXX)
add_library(DeskCamVideo SHARED Video.cpp)
target_compile_features(DeskCamVideo PRIVATE cxx_std_17)
target_compile_definitions(DeskCamVideo PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(DeskCamVideo PRIVATE /W4 /EHsc)
target_link_libraries(DeskCamVideo PRIVATE mfplat mfuuid mf ole32 wmcodecdspuuid d3d11 dxgi)
add_executable(DeskCamVideoTest VideoTest.cpp)
target_compile_features(DeskCamVideoTest PRIVATE cxx_std_17)
target_link_libraries(DeskCamVideoTest PRIVATE DeskCamVideo)
